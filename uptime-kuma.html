System Admin Project 2 Documentation

Docker engine install

1.   sudo apt update && sudo apt install -y ca-certificates curl
2.   sudo install -m 0755 -d /etc/apt/keyrings
3. sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg \  -o /etc/apt/keyrings/docker.asc
4. echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] \  https://download.docker.com/linux/ubuntu $(. /etc/os-release && echo $VERSION_CODENAME) stable" \  | sudo tee /etc/apt/sources.list.d/docker.list
5. sudo apt update
6.   sudo apt install -y docker-ce docker-ce-cli containerd.io \  docker-buildx-plugin docker-compose-plugin

Post-install checks

1. sudo systemctl enable --now docker
2. systemctl is-enabled docker
3. sudo docker run --rm hello-world
4. docker compose version
5. sudo usermod -aG docker $USER

Uptime Kuma

1. mkdir uptime-kuma
2. cd uptime-kuma
3. nano compose.yaml
4. content: services:
  uptime-kuma:
    image: louislam/uptime-kuma:2
    restart: unless-stopped
    volumes:
      - ./data:/app/data
    ports:
      # <Host Port>:<Container Port>
      - "3001:3001"
5. docker compose up -d
6. docker compose ps

SSH configuration

on Personal computer 
1. ssh-keygen -t ed25519 (generate key)
2. ssh-copy-id vm@ip address (copy key to vm)
3. create passphrase
4. sudo systemctl reload ssh
5. i can now ssh into vm without needing password, only passphrase 
6. $ sudo cat 50-cloud-init.conf
PasswordAuthentication no

Disable password

1. sudo nano /etc/ssh/sshd_config
2. find the line that says "#PasswordAuthentication yes" and delete the "#" and change yes to no

firewall:

1. sudo ufw allow OpenSSH
2. sudo ufw allow 3001/tcp (application port)
3. sudo ufw enable
4. $ sudo ufw status

auto updates:

1. sudo apt install -y unattended-upgrades
2. $ cat /etc/apt/apt.conf.d/20auto-upgrades
APT::Periodic::Update-Package-Lists "1";
APT::Periodic::Unattended-Upgrade "1";

backups:

1. mkdir backups (from home directory)
2. nano backup.sh (within uptime-kuma directory)

#!/bin/bash
set -euo pipefail

APP_DIR="$HOME/uptime-kuma"
BACKUP_DIR="$HOME/backups"
KEEP=7

mkdir -p "$BACKUP_DIR"
cd "$APP_DIR"

docker compose stop

STAMP=$(date +%Y%m%d-%H%M%S)
sudo tar -czf "$BACKUP_DIR/uptime-kuma-$STAMP.tar.gz" -C "$APP_DIR" data

docker compose start

ls -t "$BACKUP_DIR"/uptime-kuma-*.tar.gz | tail -n +$((KEEP + 1)) | xargs -r rm --

3. chmod +x backup.sh
4. ./backup.sh (Twice)

restore

1. created a monitor named RESTORE-TEST
2. docker compose down (stop the app) 
3. sudo rm -rf data (delete data)
4. NEWEST=$(ls -t ~/backups/uptime-kuma-*.tar.gz | head -1)
5. sudo tar -xzf "$NEWEST" -C ~/uptime-kuma
6. docker compose up -d
